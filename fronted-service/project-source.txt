package org.example.frontendservice.config;

import jakarta.servlet.http.HttpSession;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.ControllerAdvice;
import org.springframework.web.bind.annotation.ModelAttribute;

@ControllerAdvice
public class GlobalModelAdvice {
    @ModelAttribute
    public void auth(HttpSession session, Model model) {
        model.addAttribute("isAuthenticated", session.getAttribute("token") != null);
        Object roles = session.getAttribute("roles");
        model.addAttribute("isAdmin", roles != null && roles.toString().contains("ROLE_ADMIN"));
    }
}

package org.example.frontendservice.controller;

import jakarta.servlet.http.HttpSession;
import org.example.frontendservice.dto.CategoryDto;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.*;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.client.RestTemplate;

import java.util.Arrays;
import java.util.Map;

@Controller
@RequestMapping("/admin/categories")
public class AdminCategoryController {

    private final RestTemplate restTemplate;
    @Value("${gateway.url}")
    private String gatewayUrl;

    public AdminCategoryController(RestTemplate restTemplate) { this.restTemplate = restTemplate; }

    @GetMapping
    public String list(HttpSession session, Model model) {
        String token = (String) session.getAttribute("token");
        if (token == null) return "redirect:/login";
        try {
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);
            ResponseEntity<CategoryDto[]> response = restTemplate.exchange(
                    gatewayUrl + "/api/categories", HttpMethod.GET, new HttpEntity<>(headers), CategoryDto[].class);
            model.addAttribute("categories", Arrays.asList(response.getBody()));
        } catch (Exception e) {
            model.addAttribute("error", "Ошибка загрузки");
        }
        return "admin/categories";
    }

    @PostMapping("/create")
    public String create(@RequestParam String name, HttpSession session) {
        String token = (String) session.getAttribute("token");
        try {
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);
            headers.setContentType(MediaType.APPLICATION_JSON);
            restTemplate.postForEntity(gatewayUrl + "/api/categories", new HttpEntity<>(Map.of("name", name), headers), Void.class);
        } catch (Exception ignored) {}
        return "redirect:/admin/categories";
    }

    @PostMapping("/delete/{id}")
    public String delete(@PathVariable Long id, HttpSession session) {
        String token = (String) session.getAttribute("token");
        try {
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);
            restTemplate.exchange(gatewayUrl + "/api/categories/" + id, HttpMethod.DELETE, new HttpEntity<>(headers), Void.class);
        } catch (Exception ignored) {}
        return "redirect:/admin/categories";
    }
}

package org.example.frontendservice.controller;

import jakarta.servlet.http.HttpSession;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.*;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.client.RestTemplate;

import java.util.Map;

@Controller
@RequestMapping("/admin/components")
public class AdminComponentController {

    private final RestTemplate restTemplate;
    @Value("${gateway.url}")
    private String gatewayUrl;

    public AdminComponentController(RestTemplate restTemplate) { this.restTemplate = restTemplate; }

    @GetMapping
    public String list(@RequestParam(defaultValue = "0") int page,
                       @RequestParam(defaultValue = "10") int size,
                       @RequestParam(required = false) String search,
                       HttpSession session, Model model) {
        String token = (String) session.getAttribute("token");
        if (token == null) return "redirect:/login";
        try {
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);

            String url = gatewayUrl + "/api/components?page=" + page + "&size=" + size;
            if (search != null && !search.isEmpty()) url += "&search=" + search;

            ResponseEntity<Map> response = restTemplate.exchange(url, HttpMethod.GET, new HttpEntity<>(headers), Map.class);

            if (response.getBody() != null) {
                model.addAttribute("components", response.getBody().get("content"));
                model.addAttribute("totalPages", ((Number) response.getBody().get("totalPages")).intValue());
                model.addAttribute("currentPage", page);
            }
            model.addAttribute("search", search);
        } catch (Exception e) {
            model.addAttribute("error", "Ошибка загрузки");
        }
        return "admin/components";
    }

    @PostMapping("/delete/{id}")
    public String delete(@PathVariable Long id, HttpSession session) {
        String token = (String) session.getAttribute("token");
        try {
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);
            restTemplate.exchange(gatewayUrl + "/api/components/" + id, HttpMethod.DELETE, new HttpEntity<>(headers), Void.class);
        } catch (Exception ignored) {}
        return "redirect:/admin/components";
    }

    @PostMapping("/create")
    public String create(@RequestParam String name, @RequestParam String manufacturer,
                         @RequestParam java.math.BigDecimal price, @RequestParam Integer stockQuantity,
                         HttpSession session) {
        String token = (String) session.getAttribute("token");
        try {
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);
            headers.setContentType(MediaType.APPLICATION_JSON);

            // Предполагаем, что category.id = 1 (CPU) по умолчанию для теста
            Map<String, Object> body = Map.of(
                    "name", name, "manufacturer", manufacturer, "price", price,
                    "stockQuantity", stockQuantity, "category", Map.of("id", 1)
            );
            restTemplate.postForEntity(gatewayUrl + "/api/components", new HttpEntity<>(body, headers), Void.class);
        } catch (Exception ignored) {}
        return "redirect:/admin/components";
    }
}

package org.example.frontendservice.controller;

import jakarta.servlet.http.HttpSession;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.*;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.client.RestTemplate;

import java.util.List;
import java.util.Map;

@Controller
public class AdminController {
    private final RestTemplate restTemplate;
    @Value("${gateway.url}")
    private String gatewayUrl;

    public AdminController(RestTemplate restTemplate) { this.restTemplate = restTemplate; }

    // Главная страница админки (3 карточки)
    @GetMapping("/admin")
    public String adminPage(HttpSession session, Model model) {
        String token = (String) session.getAttribute("token");
        if (token == null) return "redirect:/login";

        Object roles = session.getAttribute("roles");
        if (roles == null || !roles.toString().contains("ROLE_ADMIN")) {
            return "redirect:/catalog";
        }

        return "admin/index"; // Отдаем страницу с карточками
    }

    // Удаление пользователя
    @PostMapping("/admin/users/{id}/delete")
    public String deleteUser(@PathVariable Long id, HttpSession session, Model model) {
        String token = (String) session.getAttribute("token");
        if (token == null) return "redirect:/login";

        try {
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);
            restTemplate.exchange(gatewayUrl + "/api/admin/users/" + id, HttpMethod.DELETE, new HttpEntity<>(headers), Void.class);
            return "redirect:/admin/users?success";
        } catch (Exception e) {
            return "redirect:/admin/users?error=" + e.getMessage();
        }
    }

    // Смена роли
    @PostMapping("/admin/users/{id}/role")
    public String changeRole(@PathVariable Long id, @RequestParam String role, HttpSession session) {
        String token = (String) session.getAttribute("token");
        if (token == null) return "redirect:/login";
        try {
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);
            headers.setContentType(MediaType.APPLICATION_JSON);
            String fullRole = role.startsWith("ROLE_") ? role : "ROLE_" + role;
            Map<String, Object> body = Map.of("roles", List.of(fullRole));
            restTemplate.exchange(gatewayUrl + "/api/admin/users/" + id + "/roles", HttpMethod.PUT, new HttpEntity<>(body, headers), Void.class);
            return "redirect:/admin/users?success";
        } catch (Exception e) {
            return "redirect:/admin/users?error=" + e.getMessage();
        }
    }
}

package org.example.frontendservice.controller;

import jakarta.servlet.http.HttpSession;
import org.example.frontendservice.dto.AuthRequest;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.*;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.client.RestTemplate;

import java.util.Map;

@Controller
public class AuthController {

    private final RestTemplate restTemplate;

    @Value("${gateway.url}")
    private String gatewayUrl;

    public AuthController(RestTemplate restTemplate) {
        this.restTemplate = restTemplate;
    }

    @GetMapping("/")
    public String index(HttpSession session, Model model) {
        if (session.getAttribute("token") != null) {
            model.addAttribute("username", session.getAttribute("username"));
            Object roles = session.getAttribute("roles");
            model.addAttribute("isAdmin", roles != null && roles.toString().contains("ROLE_ADMIN"));
        }
        return "index";
    }

    @GetMapping("/login")
    public String loginPage() {
        return "login";
    }

    @PostMapping("/login")
    public String login(@RequestParam String username,
                        @RequestParam String password,
                        HttpSession session,
                        Model model) {
        try {
            AuthRequest request = new AuthRequest(username, password);
            ResponseEntity<Map> response = restTemplate.postForEntity(
                    gatewayUrl + "/api/auth/login",
                    request,
                    Map.class
            );

            if (response.getBody() != null) {
                session.setAttribute("token", response.getBody().get("token"));
                session.setAttribute("username", response.getBody().get("username"));
                session.setAttribute("roles", response.getBody().get("roles"));
                return "redirect:/catalog";
            }
            model.addAttribute("error", "Неверный ответ от сервера");
        } catch (Exception e) {
            model.addAttribute("error", "Неверный логин или пароль");
        }
        return "login";
    }

    @GetMapping("/register")
    public String registerPage() {
        return "register";
    }

    @PostMapping("/register")
    public String register(@RequestParam String username,
                           @RequestParam String email,
                           @RequestParam String password,
                           Model model) {
        try {
            Map<String, String> body = Map.of(
                    "username", username,
                    "email", email,
                    "password", password
            );
            restTemplate.postForEntity(
                    gatewayUrl + "/api/auth/register",
                    body,
                    Map.class
            );
            return "redirect:/login?registered";
        } catch (Exception e) {
            model.addAttribute("error", "Ошибка регистрации: " + e.getMessage());
            return "register";
        }
    }

    @GetMapping("/logout")
    public String logout(HttpSession session) {
        session.invalidate();
        return "redirect:/login";
    }
}

package org.example.frontendservice.controller;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.datatype.jsr310.JavaTimeModule;
import jakarta.servlet.http.HttpSession;
import org.example.frontendservice.dto.ProductDto;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.*;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.client.RestTemplate;

import java.util.List;
import java.util.Map;

@Controller
public class BuilderController {

    private final RestTemplate restTemplate;

    @Value("${gateway.url}")
    private String gatewayUrl;

    public BuilderController(RestTemplate restTemplate) {
        this.restTemplate = restTemplate;
    }

    @GetMapping("/builder")
    public String builderPage(HttpSession session, Model model) {
        String token = (String) session.getAttribute("token");
        if (token == null) return "redirect:/login";

        try {
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);

            ResponseEntity<Map> response = restTemplate.exchange(
                    gatewayUrl + "/api/components?size=100",
                    HttpMethod.GET, new HttpEntity<>(headers), Map.class);

            if (response.getBody() != null && response.getBody().containsKey("content")) {
                List<Map<String, Object>> content = (List<Map<String, Object>>) response.getBody().get("content");
                ObjectMapper mapper = new ObjectMapper();
                mapper.registerModule(new JavaTimeModule());
                List<ProductDto> components = content.stream()
                        .map(item -> mapper.convertValue(item, ProductDto.class))
                        .toList();
                model.addAttribute("components", components);
            }
            model.addAttribute("username", session.getAttribute("username"));
        } catch (Exception e) {
            e.printStackTrace();
            model.addAttribute("error", "Ошибка загрузки компонентов: " + e.getMessage());
        }
        return "builder";
    }

    @PostMapping("/builder/save")
    public String saveBuild(@RequestParam Long componentId,
                            @RequestParam(defaultValue = "1") Integer quantity,
                            @RequestParam String name,
                            HttpSession session) {
        String token = (String) session.getAttribute("token");
        if (token == null) return "redirect:/login";

        try {
            Long userId = extractUserIdFromToken(token);

            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);
            headers.setContentType(MediaType.APPLICATION_JSON);

            Map<String, Object> buildRequest = Map.of(
                    "userId", userId,
                    "name", name,
                    "items", List.of(Map.of("componentId", componentId, "quantity", quantity))
            );

            restTemplate.exchange(gatewayUrl + "/api/builds", HttpMethod.POST, new HttpEntity<>(buildRequest, headers), Void.class);
            return "redirect:/builds?success";
        } catch (Exception e) {
            return "redirect:/builder?error=" + e.getMessage();
        }
    }

    private Long extractUserIdFromToken(String token) {
        String[] parts = token.split("\\.");
        String payload = new String(java.util.Base64.getUrlDecoder().decode(parts[1]));
        java.util.regex.Matcher m = java.util.regex.Pattern.compile("\"userId\":(\\d+)").matcher(payload);
        if (m.find()) return Long.parseLong(m.group(1));
        return 1L;
    }
}

package org.example.frontendservice.controller;

import jakarta.servlet.http.HttpSession;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.*;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.client.RestTemplate;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;

@Controller
@RequestMapping("/builds")
public class BuildsController {

    private final RestTemplate restTemplate;
    @Value("${gateway.url}")
    private String gatewayUrl;

    public BuildsController(RestTemplate restTemplate) { this.restTemplate = restTemplate; }

    @GetMapping
    public String listBuilds(HttpSession session, Model model) {
        String token = (String) session.getAttribute("token");
        if (token == null) return "redirect:/login";

        try {
            Long userId = extractUserIdFromToken(token);
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);

            ResponseEntity<List> response = restTemplate.exchange(
                    gatewayUrl + "/api/builds?userId=" + userId,
                    HttpMethod.GET, new HttpEntity<>(headers), List.class);

            model.addAttribute("builds", response.getBody());
        } catch (Exception e) {
            model.addAttribute("error", "Ошибка загрузки сборок");
        }
        return "builds";
    }

    @PostMapping("/checkout/{id}")
    public String checkoutBuild(@PathVariable Long id, HttpSession session) {
        String token = (String) session.getAttribute("token");
        if (token == null) return "redirect:/login";

        try {
            Long userId = extractUserIdFromToken(token);
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);

            ResponseEntity<Map> buildResponse = restTemplate.exchange(
                    gatewayUrl + "/api/builds/" + id,
                    HttpMethod.GET, new HttpEntity<>(headers), Map.class);
            Map<String, Object> build = buildResponse.getBody();

            List<Map<String, Object>> buildItems = (List<Map<String, Object>>) build.get("items");
            List<Map<String, Object>> orderItems = new ArrayList<>();
            for (Map<String, Object> item : buildItems) {
                Map<String, Object> component = (Map<String, Object>) item.get("component");
                orderItems.add(Map.of(
                        "componentId", component.get("id"),
                        "quantity", item.get("quantity")
                ));
            }

            Map<String, Object> orderBody = Map.of("userId", userId, "items", orderItems);
            headers.setContentType(MediaType.APPLICATION_JSON);
            restTemplate.exchange(gatewayUrl + "/api/orders", HttpMethod.POST, new HttpEntity<>(orderBody, headers), Void.class);

            // Удаляем сборку после оформления
            restTemplate.exchange(gatewayUrl + "/api/builds/" + id, HttpMethod.DELETE, new HttpEntity<>(headers), Void.class);

            return "redirect:/orders?success";
        } catch (Exception e) {
            return "redirect:/builds?error=" + e.getMessage();
        }
    }

    private Long extractUserIdFromToken(String token) {
        String[] parts = token.split("\\.");
        String payload = new String(java.util.Base64.getUrlDecoder().decode(parts[1]));
        java.util.regex.Matcher m = java.util.regex.Pattern.compile("\"userId\":(\\d+)").matcher(payload);
        if (m.find()) return Long.parseLong(m.group(1));
        return 1L;
    }
}

package org.example.frontendservice.controller;

import com.fasterxml.jackson.databind.ObjectMapper;
import jakarta.servlet.http.HttpSession;
import org.example.frontendservice.dto.ProductDto;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.*;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.client.RestTemplate;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;

@Controller
public class CatalogController {

    private final RestTemplate restTemplate;

    @Value("${gateway.url}")
    private String gatewayUrl;

    public CatalogController(RestTemplate restTemplate) {
        this.restTemplate = restTemplate;
    }

    @GetMapping("/catalog")
    public String catalog(HttpSession session, Model model) {
        String token = (String) session.getAttribute("token");
        if (token == null) return "redirect:/login";

        try {
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);

            // Запрашиваем компоненты (с параметром size=100, чтобы получить все)
            ResponseEntity<Map> response = restTemplate.exchange(
                    gatewayUrl + "/api/components?size=100",
                    HttpMethod.GET,
                    new HttpEntity<>(headers),
                    Map.class
            );

            List<ProductDto> products = new ArrayList<>();
            if (response.getBody() != null && response.getBody().containsKey("content")) {
                // Достаем список из поля content
                List<Map<String, Object>> content = (List<Map<String, Object>>) response.getBody().get("content");
                ObjectMapper mapper = new ObjectMapper();
                mapper.registerModule(new com.fasterxml.jackson.datatype.jsr310.JavaTimeModule()); // <--- ВОТ ЭТО
                for (Map<String, Object> item : content) {
                    products.add(mapper.convertValue(item, ProductDto.class));
                }
            }

            model.addAttribute("products", products);
            model.addAttribute("username", session.getAttribute("username"));

            Object roles = session.getAttribute("roles");
            model.addAttribute("isAdmin", roles != null && roles.toString().contains("ROLE_ADMIN"));

        } catch (Exception e) {
            e.printStackTrace(); // Чтобы видеть ошибку в консоли, если что-то пойдет не так
            model.addAttribute("error", "Ошибка загрузки каталога: " + e.getMessage());
        }
        return "catalog";
    }
}

    @GetMapping("/orders")
    public String ordersPage(HttpSession session, Model model) {

        if (token == null) return "redirect:/login";

        try {
            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);
            ResponseEntity<OrderDto[]> response = restTemplate.exchange(

            model.addAttribute("orders", Arrays.asList(response.getBody()));
            model.addAttribute("isAdmin", session.getAttribute("roles") != null && session.getAttribute("roles").toString().contains("ROLE_ADMIN"));
        } catch (Exception e) {
            model.addAttribute("error", "Ошибка загрузки заказов");
        }
        return "orders";
    }

    @PostMapping("/orders/create")
    public String createOrder(@RequestParam Long componentId, @RequestParam Integer quantity, HttpSession session) {
        String token = (String) session.getAttribute("token");
        if (token == null) return "redirect:/login";

        try {
            // ВМЕСТО 1 ДОСТАЕМ ID ИЗ ТОКЕНА
            Long userId = extractUserIdFromToken(token);

            HttpHeaders headers = new HttpHeaders();
            headers.set("Authorization", "Bearer " + token);
            headers.setContentType(MediaType.APPLICATION_JSON);

            Map<String, Object> body = Map.of(
                    "userId", userId, // <-- ИСПОЛЬЗУЕМ РЕАЛЬНЫЙ ID
                    "items", List.of(Map.of("componentId", componentId, "quantity", quantity))
            );
            restTemplate.exchange(gatewayUrl + "/api/orders", HttpMethod.POST, new HttpEntity<>(body, headers), Void.class);
            return "redirect:/orders?success";
        } catch (Exception e) {
            return "redirect:/catalog?error";
        }
    }

    private Long extractUserIdFromToken(String token) {
        // Простейший парсинг JWT (без библиотек) — достаем payload
        String[] parts = token.split("\\.");
        String payload = new String(java.util.Base64.getUrlDecoder().decode(parts[1]));
        // payload выглядит как {"sub":"admin","userId":1,"roles":["ROLE_ADMIN"],...}
        java.util.regex.Matcher m = java.util.regex.Pattern.compile("\"userId\":(\\d+)").matcher(payload);
        if (m.find()) return Long.parseLong(m.group(1));
        return 1L; // Fallback
    }
}
